cmake_minimum_required(VERSION 3.24)
project(linux_sdk_composition LANGUAGES C CXX ASM)
find_package(rexglue CONFIG REQUIRED)
if(NOT TARGET rex::runtime)
  message(FATAL_ERROR "An existing matching SDK installation exporting rex::runtime is required")
endif()
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  message(FATAL_ERROR "Clang or GCC required")
endif()
if(NOT PROBE_OPT STREQUAL "0" AND NOT PROBE_OPT STREQUAL "2")
  message(FATAL_ERROR "Set PROBE_OPT=0 or 2")
endif()
option(ORDINARY_ONLY "Exclude all native signal capture and hardware faults" OFF)
option(TERMINAL_DIAGNOSTICS "Fixture-only ordinary terminal witnesses" OFF)
if(TERMINAL_DIAGNOSTICS AND ORDINARY_ONLY)
  message(FATAL_ERROR "Terminal diagnostics require the real Linux transport")
endif()
add_executable(sdk_composition sdk_composition.cpp)
target_compile_features(sdk_composition PRIVATE cxx_std_23)
target_compile_options(sdk_composition PRIVATE -O${PROBE_OPT} -Wall -Wextra -Werror
  -fno-fast-math -frounding-math)
target_link_libraries(sdk_composition PRIVATE rex::runtime)
if(TERMINAL_DIAGNOSTICS)
  target_compile_definitions(sdk_composition PRIVATE LSC_TERMINAL_DIAGNOSTICS)
endif()
if(ORDINARY_ONLY)
  target_compile_definitions(sdk_composition PRIVATE LSC_ORDINARY_ONLY)
else()
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
    message(FATAL_ERROR "Native composition requires Linux x86-64")
  endif()
  target_sources(sdk_composition PRIVATE transport.c access.S)
  set_property(TARGET sdk_composition PROPERTY C_STANDARD 11)
  target_compile_options(sdk_composition PRIVATE -pthread)
  target_link_libraries(sdk_composition PRIVATE pthread)
endif()
