cmake_minimum_required(VERSION 3.24)
# Linux Runtime Bootstrap (LRB): opt-in, isolated, never part of the Windows
# release build. Consumes an EXISTING installed SDK; never builds or patches it.
project(linux_runtime_bootstrap LANGUAGES C CXX)
find_package(rexglue CONFIG REQUIRED)
if(NOT TARGET rex::runtime)
  message(FATAL_ERROR "An existing matching SDK installation exporting rex::runtime is required")
endif()
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  message(FATAL_ERROR "Clang or GCC required")
endif()
if(NOT LRB_OPT STREQUAL "0" AND NOT LRB_OPT STREQUAL "2")
  message(FATAL_ERROR "Set LRB_OPT=0 or 2")
endif()
option(LRB_ORDINARY_ONLY "No signal owner or hardware faults (any platform)" OFF)
add_executable(lrb_tests lrb_tests.cpp lrb_bootstrap.cpp)
target_compile_features(lrb_tests PRIVATE cxx_std_23)
target_compile_options(lrb_tests PRIVATE -O${LRB_OPT} -Wall -Wextra -Werror
  -fno-fast-math -frounding-math)
target_link_libraries(lrb_tests PRIVATE rex::runtime)
if(WIN32)
  # MSVC CRT marks ISO getenv/fopen deprecated; the code uses only ISO forms.
  target_compile_definitions(lrb_tests PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
if(LRB_ORDINARY_ONLY)
  target_compile_definitions(lrb_tests PRIVATE LRB_ORDINARY_ONLY)
else()
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
    message(FATAL_ERROR "The native LRB owner requires Linux x86-64")
  endif()
  enable_language(ASM)
  target_sources(lrb_tests PRIVATE lrb_owner.c lrb_access.S)
  # Test executable: compile the negative wrong-site hook into the owner copy.
  target_compile_definitions(lrb_tests PRIVATE LRB_TEST)
  set_property(TARGET lrb_tests PROPERTY C_STANDARD 11)
  target_compile_options(lrb_tests PRIVATE -pthread)
  target_link_libraries(lrb_tests PRIVATE pthread)
endif()
