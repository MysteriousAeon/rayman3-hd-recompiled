cmake_minimum_required(VERSION 3.24)
# Linux Runtime Generated Dispatch (LRGD): opt-in, isolated, never part of the
# Windows release build. Consumes an EXISTING installed SDK; never builds or
# patches it. The selected generated functions come from the PRIVATE, unchanged
# LGE extract (LGE_GENERATED_DIR: lge_generated.cpp + verbatim r3hd_pch.h).
project(linux_runtime_dispatch LANGUAGES C CXX)
find_package(rexglue CONFIG REQUIRED)
if(NOT TARGET rex::runtime)
  message(FATAL_ERROR "An existing matching SDK installation exporting rex::runtime is required")
endif()
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  message(FATAL_ERROR "Clang or GCC required")
endif()
if(NOT LRGD_OPT STREQUAL "0" AND NOT LRGD_OPT STREQUAL "2")
  message(FATAL_ERROR "Set LRGD_OPT=0 or 2")
endif()
if(NOT LGE_GENERATED_DIR OR NOT EXISTS "${LGE_GENERATED_DIR}/lge_generated.cpp"
   OR NOT EXISTS "${LGE_GENERATED_DIR}/r3hd_pch.h")
  message(FATAL_ERROR "LGE_GENERATED_DIR must hold the private lge_generated.cpp and r3hd_pch.h")
endif()
if(EXISTS "${LGE_GENERATED_DIR}/m2_trace.h")
  message(FATAL_ERROR "The private directory must not shadow the LGE m2_trace.h shim")
endif()
option(LRB_ORDINARY_ONLY "No signal owner or hardware faults (any platform)" OFF)
set(LRB_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../linux_runtime_bootstrap")
set(LGE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../linux_guest_entry")
set(LWR_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../linux_write_watch_restart")
# Exactly the LGE flags, so the generated object is the audited LGE object.
set(LRGD_STRICT -O${LRGD_OPT} -Wall -Wextra -Werror -fno-fast-math -frounding-math)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
  list(APPEND LRGD_STRICT $<$<COMPILE_LANGUAGE:CXX>:-msse4.1>)
endif()

add_library(lge_generated OBJECT "${LGE_GENERATED_DIR}/lge_generated.cpp")
target_compile_features(lge_generated PRIVATE cxx_std_23)
target_include_directories(lge_generated BEFORE PRIVATE "${LGE_DIR}" "${LGE_GENERATED_DIR}")
target_link_libraries(lge_generated PRIVATE rex::runtime)
target_compile_options(lge_generated PRIVATE ${LRGD_STRICT})
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  set(LGE_PORTABLE_DEFINE_DEFAULT ON)
else()
  set(LGE_PORTABLE_DEFINE_DEFAULT OFF)
endif()
option(LGE_PORTABLE_DEFINE "Force-include lge_define_rex_func.h for the generated object"
       ${LGE_PORTABLE_DEFINE_DEFAULT})
if(LGE_PORTABLE_DEFINE)
  target_compile_options(lge_generated PRIVATE -include "${LGE_DIR}/lge_define_rex_func.h")
endif()

add_executable(lrgd_tests lrgd_tests.cpp "${LRB_DIR}/lrb_bootstrap.cpp" $<TARGET_OBJECTS:lge_generated>)
target_compile_features(lrgd_tests PRIVATE cxx_std_23)
target_include_directories(lrgd_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${LRB_DIR}" "${LWR_DIR}")
target_compile_options(lrgd_tests PRIVATE ${LRGD_STRICT})
target_link_libraries(lrgd_tests PRIVATE rex::runtime)
if(WIN32)
  target_compile_definitions(lrgd_tests PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
if(LRB_ORDINARY_ONLY)
  target_compile_definitions(lrgd_tests PRIVATE LRB_ORDINARY_ONLY)
else()
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
    message(FATAL_ERROR "The native LRGD owner requires Linux x86-64")
  endif()
  enable_language(ASM)
  # The audited LRB owner/transport and the audited LWR admission path, built
  # exactly as the LWR and LRSO gates build them (same per-source flags).
  set_source_files_properties("${LWR_DIR}/lwr_restart.c" PROPERTIES COMPILE_OPTIONS
                              "-fno-stack-protector;-mgeneral-regs-only")
  target_sources(lrgd_tests PRIVATE "${LRB_DIR}/lrb_owner.c" "${LRB_DIR}/lrb_access.S" "${LWR_DIR}/lwr_restart.c")
  target_compile_definitions(lrgd_tests PRIVATE LRB_TEST LRB_RESTART)
  set_property(TARGET lrgd_tests PROPERTY C_STANDARD 11)
  target_compile_options(lrgd_tests PRIVATE -pthread)
  # Eager binding: no lazy PLT resolution on any signal-context path.
  target_link_options(lrgd_tests PRIVATE -Wl,-z,now -Wl,-z,relro)
  target_link_libraries(lrgd_tests PRIVATE pthread ${CMAKE_DL_LIBS})
endif()
